cmake_minimum_required(VERSION 3.16)
project(hd485 VERSION 0.1.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
add_compile_definitions(_GNU_SOURCE)
add_compile_options(-Wall -Wextra -Wpedantic)
option(HD485_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(HD485_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(hd485_wire STATIC src/wire.c src/sha256.c)
target_include_directories(hd485_wire PUBLIC include src)
add_library(hd485 STATIC src/sdk.c)
target_link_libraries(hd485 PUBLIC hd485_wire)
add_executable(hd485d src/daemon.c src/serial.c src/maintenance.c)
target_link_libraries(hd485d PRIVATE hd485_wire Threads::Threads)
add_executable(hd485d-test src/daemon.c src/serial.c src/maintenance.c)
target_compile_definitions(hd485d-test PRIVATE HD485_TEST_IO=1)
target_link_libraries(hd485d-test PRIVATE hd485_wire Threads::Threads)
add_executable(hd485-example examples/client.c)
target_link_libraries(hd485-example PRIVATE hd485)
enable_testing()
add_executable(wire-test tests/wire_test.c)
target_link_libraries(wire-test PRIVATE hd485_wire)
add_test(NAME wire COMMAND wire-test)
install(TARGETS hd485d hd485 hd485_wire hd485-example RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES include/hd485.h DESTINATION include)
